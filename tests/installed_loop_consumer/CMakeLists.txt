cmake_minimum_required(VERSION 3.16)
project(livo_atlas_loop_consumer LANGUAGES CXX)

if(USE_CATKIN)
  find_package(catkin REQUIRED COMPONENTS livo_atlas)
else()
  find_package(livo_atlas CONFIG REQUIRED)
  if(EXPECT_LOOP AND NOT TARGET livo_atlas::loop_closure)
    message(FATAL_ERROR "The loop target was not exported")
  elseif(NOT EXPECT_LOOP AND TARGET livo_atlas::loop_closure)
    message(FATAL_ERROR "The disabled loop target was exported")
  endif()
endif()
add_executable(loop_consumer main.cc)
target_compile_features(loop_consumer PRIVATE cxx_std_17)
if(EXPECT_LOOP)
  target_compile_definitions(loop_consumer PRIVATE EXPECT_LOOP)
endif()
if(USE_CATKIN)
  target_include_directories(loop_consumer PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(loop_consumer PRIVATE ${catkin_LIBRARIES})
elseif(EXPECT_LOOP)
  target_link_libraries(loop_consumer PRIVATE livo_atlas::loop_closure)
else()
  target_link_libraries(loop_consumer PRIVATE livo_atlas::geometry)
endif()
enable_testing()
add_test(NAME loop_consumer COMMAND loop_consumer)

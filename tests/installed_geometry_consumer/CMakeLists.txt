cmake_minimum_required(VERSION 3.16)
project(installed_geometry_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LIVO_ATLAS_CONSUMER_ROS_VERSION "1" CACHE STRING "1=Noetic, 2=Humble")
add_executable(geometry_consumer main.cc)
target_compile_definitions(geometry_consumer PRIVATE
  LIVO_ATLAS_CONSUMER_ROS_VERSION=${LIVO_ATLAS_CONSUMER_ROS_VERSION})
if(LIVO_ATLAS_CONSUMER_ROS_VERSION STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS livo_atlas roscpp)
  target_include_directories(geometry_consumer PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(geometry_consumer ${catkin_LIBRARIES})
elseif(LIVO_ATLAS_CONSUMER_ROS_VERSION STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(livo_atlas REQUIRED)
  find_package(rclcpp REQUIRED)
  ament_target_dependencies(geometry_consumer livo_atlas rclcpp)
else()
  message(FATAL_ERROR "Expected Noetic or Humble message consumer")
endif()
enable_testing()
add_test(NAME geometry_consumer COMMAND geometry_consumer)

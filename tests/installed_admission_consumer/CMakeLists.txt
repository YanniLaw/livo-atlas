cmake_minimum_required(VERSION 3.16)
project(livo_atlas_admission_consumer LANGUAGES CXX)

option(EXPECT_GTSAM "Expect the pose graph" OFF)
option(EXPECT_LOOP "Expect geometric loop verification" OFF)
option(USE_CATKIN "Consume catkin's flattened library exports" OFF)
set(expect_admission OFF)
if(EXPECT_GTSAM AND EXPECT_LOOP)
  set(expect_admission ON)
endif()
if(USE_CATKIN)
  find_package(catkin REQUIRED COMPONENTS livo_atlas)
else()
  find_package(livo_atlas CONFIG REQUIRED)
endif()

function(check_optional name expected)
  if(USE_CATKIN)
    string(FIND "${catkin_LIBRARIES}" "livo_atlas_${name}" found)
    if(expected AND found EQUAL -1 OR NOT expected AND NOT found EQUAL -1)
      message(FATAL_ERROR "Incorrect catkin export for ${name}")
    endif()
  else()
    if(expected AND NOT TARGET livo_atlas::${name})
      message(FATAL_ERROR "Missing optional target ${name}")
    elseif(NOT expected AND TARGET livo_atlas::${name})
      message(FATAL_ERROR "Disabled target ${name} was exported")
    endif()
  endif()
endfunction()
check_optional(backend ${EXPECT_GTSAM})
check_optional(loop_closure ${EXPECT_LOOP})
check_optional(loop_backend ${expect_admission})
check_optional(online_backend ${expect_admission})

add_executable(admission_consumer main.cc)
target_compile_features(admission_consumer PRIVATE cxx_std_17)
if(EXPECT_GTSAM)
  target_compile_definitions(admission_consumer PRIVATE EXPECT_GTSAM)
endif()
if(EXPECT_LOOP)
  target_compile_definitions(admission_consumer PRIVATE EXPECT_LOOP)
endif()
if(expect_admission)
  target_compile_definitions(admission_consumer PRIVATE EXPECT_ADMISSION)
endif()
if(USE_CATKIN)
  target_include_directories(admission_consumer PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(admission_consumer PRIVATE ${catkin_LIBRARIES})
else()
  target_link_libraries(admission_consumer PRIVATE livo_atlas::core livo_atlas::keyframes)
  if(EXPECT_GTSAM)
    target_link_libraries(admission_consumer PRIVATE livo_atlas::backend)
  endif()
  if(EXPECT_LOOP)
    target_link_libraries(admission_consumer PRIVATE livo_atlas::loop_closure)
  endif()
  if(expect_admission)
    target_link_libraries(admission_consumer PRIVATE livo_atlas::online_backend)
  endif()
endif()
enable_testing()
add_test(NAME admission_consumer COMMAND admission_consumer)

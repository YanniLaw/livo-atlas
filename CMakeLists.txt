cmake_minimum_required(VERSION 3.16)
project(livo_atlas VERSION 0.0.0 LANGUAGES C CXX)

set(livo_atlas_default_ros_version "1")
if("$ENV{ROS_VERSION}" MATCHES "^[12]$")
  set(livo_atlas_default_ros_version "$ENV{ROS_VERSION}")
endif()
set(LIVO_ATLAS_ROS_VERSION "${livo_atlas_default_ros_version}" CACHE STRING
  "ROS integration: 0 (none), 1 (Noetic), 2 (Humble)")
set_property(CACHE LIVO_ATLAS_ROS_VERSION PROPERTY STRINGS 0 1 2)
if(NOT LIVO_ATLAS_ROS_VERSION MATCHES "^[012]$")
  message(FATAL_ERROR "LIVO_ATLAS_ROS_VERSION must be 0, 1 or 2")
endif()
if(LIVO_ATLAS_ROS_VERSION STREQUAL "2" AND NOT "$ENV{ROS_DISTRO}" STREQUAL "humble")
  message(FATAL_ERROR "ROS2 integration requires a sourced Humble environment")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(LIVO_ATLAS_USE_OPENMP "Enable estimator parallel loops" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)

if(LIVO_ATLAS_ROS_VERSION STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS geometry_msgs nav_msgs sensor_msgs
    roscpp rospy std_msgs pcl_ros tf message_generation eigen_conversions
    cv_bridge image_transport visualization_msgs)
endif()
if(LIVO_ATLAS_ROS_VERSION STREQUAL "2")
  find_package(ament_cmake REQUIRED)
endif()
include(cmake/core.cmake)
if(LIVO_ATLAS_ROS_VERSION STREQUAL "1")
  include(cmake/ros1.cmake)
  set(LIVO_ATLAS_BUILD_TESTS ${CATKIN_ENABLE_TESTING})
else()
  if(LIVO_ATLAS_ROS_VERSION STREQUAL "2")
    include(cmake/ros2.cmake)
  endif()
  include(CTest)
  set(LIVO_ATLAS_BUILD_TESTS ${BUILD_TESTING})
endif()
if(NOT LIVO_ATLAS_ROS_VERSION STREQUAL "0")
  include(cmake/node.cmake)
endif()
if(LIVO_ATLAS_BUILD_TESTS)
  include(cmake/tests.cmake)
endif()

if(LIVO_ATLAS_ROS_VERSION STREQUAL "2")
  ament_package(CONFIG_EXTRAS ${PROJECT_BINARY_DIR}/ros2_dependencies.cmake)
endif()

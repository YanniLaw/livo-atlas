cmake_minimum_required(VERSION 3.16)
project(livo_atlas VERSION 0.0.0 LANGUAGES C CXX)

set(LIVO_ATLAS_ROS_VERSION "1" CACHE STRING "ROS integration: 0 (none), 1 (Noetic)")
set_property(CACHE LIVO_ATLAS_ROS_VERSION PROPERTY STRINGS 0 1)
if(NOT LIVO_ATLAS_ROS_VERSION MATCHES "^[01]$")
  message(FATAL_ERROR "LIVO_ATLAS_ROS_VERSION must be 0 or 1; ROS2 Humble is planned for P4")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(LIVO_ATLAS_USE_OPENMP "Enable estimator parallel loops" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)

if(LIVO_ATLAS_ROS_VERSION STREQUAL "1")
  find_package(catkin REQUIRED COMPONENTS geometry_msgs nav_msgs sensor_msgs
    roscpp rospy std_msgs pcl_ros tf message_generation eigen_conversions
    cv_bridge image_transport visualization_msgs)
endif()
include(cmake/core.cmake)
if(LIVO_ATLAS_ROS_VERSION STREQUAL "1")
  include(cmake/ros1.cmake)
  set(LIVO_ATLAS_BUILD_TESTS ${CATKIN_ENABLE_TESTING})
else()
  include(CTest)
  set(LIVO_ATLAS_BUILD_TESTS ${BUILD_TESTING})
endif()
if(LIVO_ATLAS_BUILD_TESTS)
  include(cmake/tests.cmake)
endif()
